14 - TOTAL HORAS APONTADAS = 'glpi_tickets'[Total_Tempo_Trabalhado_Task_Coluna]

	Total_Tempo_Trabalhado_Task_Coluna = 
VAR TotalSegundos = glpi_tickets[actiontime]  // Usa o valor da linha atual
VAR HorasTotais = QUOTIENT(TotalSegundos, 3600)  
VAR MinutosTotais = QUOTIENT(MOD(TotalSegundos, 3600), 60)  
VAR SegundosTotais = MOD(TotalSegundos, 60)  
VAR AjusteHoras = HorasTotais + QUOTIENT(MinutosTotais, 60)  
VAR AjusteMinutos = MOD(MinutosTotais, 60)  

RETURN 
    IF(TotalSegundos = 0, 
        "",  
        FORMAT(AjusteHoras, "00") & ":" & FORMAT(AjusteMinutos, "00") & ":" & FORMAT(SegundosTotais, "00")  
    )

15 - HORAS APONTADAS = glpi_tickettasks'[_HorasTrabalhadas_Tasks]

	_HorasTrabalhadas_Tasks = 

VAR TotalSegundos = [_action_time_second_tasks]

VAR Horas = INT(TotalSegundos / 3600)

VAR Minutos = 
INT(
    MOD(TotalSegundos,3600) / 60
)

VAR Segundos = 
MOD(TotalSegundos,60)

RETURN

IF(
    TotalSegundos = 0,
    BLANK(),
    FORMAT(Horas,"00") & ":" &
    FORMAT(Minutos,"00") & ":" &
    FORMAT(Segundos,"00")
)
